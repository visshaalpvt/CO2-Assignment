SELECTION_SORT(A)
 for i = 0 to n-2:
   minIndex = i
   for j = i+1 to n-1:
     comparisons += 1
     if A[j] < A[minIndex]: minIndex = j
   if minIndex != i: swap A[i], A[minIndex]; swaps += 1
 return A

BUBBLE_SORT(A)
 for end = n-1 down to 1:
   swapped = false
   for j = 0 to end-1:
     comparisons += 1
     if A[j] > A[j+1]: swap; swaps += 1; swapped = true
   if not swapped: break
 return A

INSERTION_SORT(A)
 for i = 1 to n-1:
   key = A[i]; j = i-1
   while j >= 0:
     comparisons += 1
     if A[j] > key: A[j+1] = A[j]; shifts += 1; j -= 1
     else: break
   A[j+1] = key
 return A
